{{/* Include a Markdown asset (path relative to assets/, e.g. includes/FOO.md). Parent docs are mounted in hugo.yaml: ../ARCHITECTURE.md → assets/includes/ARCHITECTURE.md Usage: {{% include path="includes/ARCHITECTURE.md" %}} Relative monorepo links that work on GitHub are rewritten for the site: ./SERVER.md → docs/server/ ./ARCHITECTURE.md → docs/architecture/ ../path/in/repo → https://github.com/gomuks/gomuks/blob/main/path/in/repo */}} {{- $path := .Get "path" | default (.Get 0) -}} {{- if not $path -}} {{- errorf "shortcode include: missing path attribute" -}} {{- end -}} {{- $res := resources.Get $path -}} {{- if not $res -}} {{- errorf "shortcode include: asset not found: %s" $path -}} {{- end -}} {{- $raw := $res.Content -}} {{- /* Drop a leading ATX H1 so Docsy page title is not duplicated. */ -}} {{- $raw = $raw | replaceRE `(?s)^#[^\n]+\n+` "" -}} {{- /* Cross-links between in-repo guides → Docsy pages (root-relative). */ -}} {{- $raw = $raw | replaceRE `\]\(\./SERVER\.md\)` (printf "](%s)" ("docs/server/" | relURL)) -}} {{- $raw = $raw | replaceRE `\]\(\./ARCHITECTURE\.md\)` (printf "](%s)" ("docs/architecture/" | relURL)) -}} {{- /* "website/" directory mentions are not navigable pages; point at docs site guide. */ -}} {{- $raw = $raw | replaceRE `\]\(\./website/?\)` (printf "](%s)" ("docs/documentation-site/" | relURL)) -}} {{- /* Other relative repo paths → GitHub blob (main). */ -}} {{- $raw = $raw | replaceRE `\]\(\.\./([^)#\s]+)\)` "](https://github.com/gomuks/gomuks/blob/main/$1)" -}} {{- $raw | .Page.RenderString -}}
